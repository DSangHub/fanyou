-- ===================================================
-- FANPULSE REAL-TIME ENGAGEMENT DATABASE SCHEMA
-- PostgreSQL 15+ / Supabase Compatible DDL
-- ===================================================

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. SUBSCRIPTION TIERS ENUM & TABLE
CREATE TYPE subscription_tier_enum AS ENUM ('FREE', 'FAN_VIP', 'ULTIMATE_PRO');

CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    avatar_url TEXT,
    tier subscription_tier_enum DEFAULT 'FREE',
    stripe_customer_id VARCHAR(255),
    points_balance INT DEFAULT 0 CHECK (points_balance >= 0),
    streak_count INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. LIVE STREAMS & EVENTS TABLE
CREATE TABLE events (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(255) NOT NULL,
    description TEXT,
    status VARCHAR(20) DEFAULT 'LIVE', -- SCHEDULED, LIVE, ENDED
    stream_url TEXT,
    started_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. REAL-TIME POLLS & OPTIONS
CREATE TABLE polls (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    event_id UUID REFERENCES events(id) ON DELETE CASCADE,
    question TEXT NOT NULL,
    min_tier_required subscription_tier_enum DEFAULT 'FREE',
    point_reward INT DEFAULT 50,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE poll_options (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    poll_id UUID REFERENCES polls(id) ON DELETE CASCADE,
    option_text VARCHAR(255) NOT NULL,
    vote_count INT DEFAULT 0
);

CREATE TABLE poll_votes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    poll_id UUID REFERENCES polls(id),
    option_id UUID REFERENCES poll_options(id),
    user_id UUID REFERENCES users(id),
    voted_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(poll_id, user_id) -- Prevents double voting
);

-- 4. GAMIFICATION TRANSACTION LOGS
CREATE TABLE point_transactions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES users(id),
    amount INT NOT NULL, -- Positive for earnings, negative for redemptions
    reason VARCHAR(100) NOT NULL, -- POLL_VOTE, REACTION, REWARD_REDEMPTION
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- High concurrency indexes for Real-time Leaderboards & Chat
CREATE INDEX idx_users_points ON users (points_balance DESC);
CREATE INDEX idx_poll_votes_poll ON poll_votes (poll_id);
